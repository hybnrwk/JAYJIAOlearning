Conditional Expectation→Markov Chain→Poisson Process→CTMC→Renewal→Brownian Motion→Martingale\boxed{ \text{Conditional Expectation} \rightarrow \text{Markov Chain} \rightarrow \text{Poisson Process} \rightarrow \text{CTMC} \rightarrow \text{Renewal} \rightarrow \text{Brownian Motion} \rightarrow \text{Martingale} }

Markov Chain→{MCMCMDP/RLHMM\text{Markov Chain} \rightarrow \begin{cases} \text{MCMC}\\ \text{MDP/RL}\\ \text{HMM} \end{cases}
